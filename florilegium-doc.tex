\directlua{florilegiummanual = dofile("docs/manual.lua")}
\directlua{
  local version, date = florilegiummanual.release()
  local y, m, d = date:match("(.+)/(.+)/(.+)")
  local months = { "January", "February", "March", "April", "May", "June", "July",
    "August", "September", "October", "November", "December" }
  tex.sprint("\string\\def\string\\floversion{" .. version .. "}")
  tex.sprint("\string\\def\string\\flodate{" .. tonumber(d) .. " " .. months[tonumber(m)] .. " " .. y .. "}")
}
